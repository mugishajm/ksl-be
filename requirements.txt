# Avoid 3.13+ unless you know wheels exist for your platform.
# This project is not supported on Python 3.14+ on Windows because core deps (NumPy/Pandas)

# NumPy + OpenCV baseline (supported on Python 3.11–3.13).

numpy==1.26.4; python_version < "3.14"
# Keep a single OpenCV wheel installed to avoid cv2/numpy ABI conflicts.
# MediaPipe already pulls OpenCV; we pin contrib build explicitly for stability.
opencv-contrib-python==4.11.0.86
# MediaPipe + TensorFlow must agree on a protobuf major version.
# Newer MediaPipe wheels work with protobuf 6/7 (required by TF 2.21+).

# TensorFlow 2.21 requires protobuf >= 6.31.1.
protobuf>=6.31.1,<8
requests>=2.31.0
pymongo>=4.8.0
